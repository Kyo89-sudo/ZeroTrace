cmake_minimum_required(VERSION 3.16)
project(ZeroTrace CXX RC)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
if(MSVC)
  add_compile_options(/W3 /utf-8)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(ENTRY_FLAGS "/ENTRY:wWinMainCRTStartup /MANIFEST:NO")
else()
  set(ENTRY_FLAGS "-municode -static")
endif()
set(LIBS comctl32 shell32 shlwapi ole32 uuid advapi32 uxtheme gdi32 gdiplus msimg32)

add_executable(ZeroTrace WIN32 src/main.cpp src/scanner.cpp src/ZeroTrace.rc)
target_link_libraries(ZeroTrace ${LIBS})
set_target_properties(ZeroTrace PROPERTIES LINK_FLAGS "${ENTRY_FLAGS}")

add_executable(StubbornApp WIN32 test/StubbornApp/stubborn.cpp test/StubbornApp/StubbornApp.rc)
target_link_libraries(StubbornApp ${LIBS})
set_target_properties(StubbornApp PROPERTIES LINK_FLAGS "${ENTRY_FLAGS}")

add_executable(selftest test/selftest.cpp src/scanner.cpp)
target_link_libraries(selftest ${LIBS})
if(NOT MSVC)
  set_target_properties(selftest PROPERTIES LINK_FLAGS "-municode -static")
endif()
